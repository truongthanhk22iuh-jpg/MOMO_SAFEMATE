<section id="faq"><div class="wrap">
  <div class="sec-head"><div class="kicker">Hỏi đáp</div><h2>Câu hỏi thường gặp</h2></div>
  <div style="max-width:820px">
    <details class="faq-item"><summary>Kết quả kiểm tra có chắc chắn 100% không?</summary><div class="faq-body">Không. Kết quả dựa trên nội dung bạn cung cấp và mang tính tham khảo để bạn biết cần xác minh thêm điều gì. Một giao dịch được đánh giá “tương đối rõ” vẫn nên được đối chiếu lại (gặp trực tiếp, kiểm tra kênh chính thức, giữ biên nhận) trước khi chuyển tiền.</div></details>
    <details class="faq-item"><summary>Nội dung tôi nhập có bị lưu lại hay gửi đi đâu không?</summary><div class="faq-body">Không. Việc phân tích chạy ngay trên trình duyệt của bạn; nội dung bạn dán vào không được gửi lên máy chủ. Lịch sử kiểm tra chỉ tồn tại trong lúc bạn mở trang — đóng trang là mất. Dù vậy, bạn vẫn nên che bớt thông tin cá nhân nhạy cảm trước khi dán, như một thói quen an toàn chung.</div></details>
    <details class="faq-item"><summary>Có cần đăng nhập hay tạo tài khoản không?</summary><div class="faq-body">Không. Bạn mở trang là dùng được ngay, không cần đăng nhập, không cần liên kết ví hay tài khoản ngân hàng.</div></details>

function fmtMoney(n){ if(!n) return '—'; return Number(n).toLocaleString('vi-VN')+'đ'; }
function extractAmount(text){
  var m = text.replace(/\./g,'').match(/(\d{4,9})\s*(đ|dong|vnd)?/i);
  if(m) return parseInt(m[1],10);
  var k = text.match(/(\d+(?:[.,]\d+)?)\s*(triệu|tr)/i); if(k) return Math.round(parseFloat(k[1].replace(',','.'))*1000000);
  var kk = text.match(/(\d+)\s*k\b/i); if(kk) return parseInt(kk[1],10)*1000;
  return null;
}
function extractDeadline(text){
  var m = text.match(/(trong \d+ phút|hôm nay|tối nay|trước \d{1,2}h|ngày \d{1,2}\/\d{1,2}|thứ [a-zà-ỹ]+ tuần này|cuối tuần này)/i);
  return m ? m[1] : null;
}
function esc(s){ return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

function analyzeData(type, amountInput, receiver, content){
  var lower = (' '+content.toLowerCase()+' ');
  var findings = [], score = 0;
  RULES.forEach(function(r){
    var hit = r.kw.filter(function(k){ return lower.indexOf(k) >= 0; });
    if(hit.length){ score += r.score; findings.push({title:r.title, reason:r.reason, quote:hit.join('”, “'), score:r.score}); }
  });
  var hasRefund = /(hoàn cọc|hoàn tiền|trả lại cọc|đổi trả|hoàn lại)/i.test(content);
  if((type==='datcoc' || /cọc/.test(content)) && !hasRefund){ score += 15; findings.push({title:'Chưa rõ điều kiện hoàn cọc / hoàn tiền', reason:'Nội dung không nhắc trường hợp nào được hoàn lại tiền.   var amountInText = extractAmount(content);
  var amount = amountInput ? parseInt(String(amountInput).replace(/[^\d]/g,''),10) : amountInText;
  if(content.trim().length < 60){ score += 20; findings.push({title:'Nội dung quá thiếu thông tin', reason:'Nội dung bạn cung cấp rất ngắn, chưa rõ số tiền, mục đích, thời hạn và điều kiện — chưa đủ căn cứ để quyết định chuyển tiền.', quote:'Độ dài nội dung: '+content.trim().length+' ký tự', score:20}); }
  if(!amount){ score += 10; findings.push({title:'Chưa xác định được số tiền', reason:'Cả form và nội dung đều không nêu rõ số tiền. Hãy hỏi lại con số chính xác và tổng chi phí cuối cùng.', quote:'Không trích xuất được số tiền', score:10}); }

  var positives = POSITIVE.filter(function(p){ return p.kw.some(function(k){ return lower.indexOf(k)>=0; }); }).map(function(p){return p.text;});

  return {type:type, amount:amount, receiver:receiver, deadline:extractDeadline(content), findings:findings, positives:positives, score:score, level:level, content:content};
}

#include <algorithm>

struct Signal {
    std::string code;    // ma dau hieu
    std::string message; // giai thich cho nguoi dung
    int weight;          // diem rui ro cong them
};

// Chuyen chuoi ve chu thuong (bo qua dau tieng Viet o muc co ban:
// chi xu ly ASCII, du dung cho tu khoa khong dau / co dau dang UTF-8 nguyen van).
static std::string toLower(const std::string& s) {
    std::string r = s;
    std::transform(r.begin(), r.end(), r.begin(),
                   [](unsigned char c) { return std::tolower(c); });
    return r;
}

static bool contains(const std::string& text, const std::string& kw) {
    return text.find(kw) != std::string::npos;
}

static bool containsAny(const std::string& text,
                        const std::vector<std::string>& kws) {
    for (const auto& k : kws)
        if (contains(text, k)) return true;
    return false;
}

// Trich xuat so tien lon nhat dang "500000" / "500.000" / "2tr" tu noi dung.
static long long extractAmount(const std::string& text) {
    long long best = 0;
    std::string digits;
    for (size_t i = 0; i <= text.size(); ++i) {
        if (i < text.size() && (std::isdigit((unsigned char)text[i]) || text[i] == '.')) {
            if (text[i] != '.') digits += text[i];
        } else {
            if (digits.size() >= 5) { // so tien VND thuong >= 10.000
                long long v = 0;
                try { v = std::stoll(digits); } catch (...) { v = 0; }
                if (v > best) best = v;
            }
            digits.clear();
        }
    }
    return best;
}

int main(int argc, char* argv[]) {
    if (argc < 2) {
        std::cout << "{\"error\":\"Thieu noi dung giao dich. "
                     "Dung: ./backend \\\"<noi dung>\\\" [so_tien]\"}\n";
        return 1;
    }
    std::string content = argv[1];
    long long formAmount = (argc >= 3) ? std::atoll(argv[2]) : 0;
    std::string t = toLower(content);

    std::vector<Signal> signals;
    int score = 0;
    auto add = [&](const std::string& code, const std::string& msg, int w) {
        signals.push_back({code, msg, w});
        score += w;
    };

    // 1) Dau hieu cuc ky nguy hiem: yeu cau OTP / mat khau
    if (containsAny(t, {"otp", "ma xac thuc", "mat khau", "password"}))
        add("OTP_REQUEST",
            "Noi dung co yeu cau cung cap ma OTP/mat khau. Khong bao gio "
            "chia se OTP hay mat khau cho bat ky ai.", 60);

    // 2) Hoi thuc chuyen tien gap
    if (containsAny(t, {"ngay", "gap", "trong 10 phut", "trong 30 phut",
                        "keo het", "het phong", "het hang", "chi con",
                        "nhanh tay", "khong thi"}))
        add("URGENCY",
            "Ben nhan tao ap luc thoi gian, thuc chuyen tien gap - dau "
            "hieu can xac minh them truoc khi chuyen.", 25);

    // 3) Yeu cau coc/dong phi truoc ma dieu kien khong ro
    bool asksPrepay = containsAny(t, {"coc", "dat coc", "dong truoc",
                                      "chuyen truoc", "ung truoc", "phi"});
    bool hasCondition = containsAny(t, {"hoan coc", "hoan tien", "hop dong",
                                        "bien nhan", "dieu kien", "cam ket"});
    if (asksPrepay && !hasCondition)
        add("PREPAY_NO_TERMS",
            "Yeu cau tra tien truoc nhung khong neu dieu kien hoan tien/"
            "hoan coc hay chung tu - can hoi ro truoc khi chuyen.", 20);

    // 4) Gia re bat thuong
    if (containsAny(t, {"re nhat", "gia soc", "giam 70", "giam 80", "giam 90",
                        "x thanh ly", "thanh ly gap"}))
        add("TOO_CHEAP",
            "Muc gia/uu dai re bat thuong so voi thi truong - can doi "
            "chieu gia va kiem tra hang that.", 15);

    // 5) Thong tin qua thieu
    long long found = extractAmount(t);
    if (content.size() < 60 && found == 0 && formAmount == 0)
        add("INSUFFICIENT_INFO",
            "Noi dung qua ngan, thieu so tien/dieu kien - chua du can cu "
            "de danh gia, can hoi them thong tin.", 15);

    // 6) Lech so tien giua form va noi dung
    if (formAmount > 0 && found > 0 && found != formAmount)
        add("AMOUNT_MISMATCH",
            "So tien trong noi dung khac so tien ban khai bao - can doi "
            "chieu lai truoc khi chuyen.", 15);

    if (score > 100) score = 100;
    std::string level = (score >= 50) ? "RUI_RO_CAO"
                        : (score >= 20) ? "CAN_XAC_MINH_THEM"
                                        : "TUONG_DOI_RO";

    std::cout << "{\n";
    std::cout << "  \"amount_found\": " << (found > 0 ? found : formAmount) << ",\n";
    std::cout << "  \"risk_score\": " << score << ",\n";
    std::cout << "  \"level\": \"" << level << "\",\n";
    std::cout << "  \"signals\": [";
    for (size_t i = 0; i < signals.size(); ++i) {
        if (i) std::cout << ",";
        std::cout << "\n    {\"code\": \"" << signals[i].code
                  << "\", \"weight\": " << signals[i].weight
                  << ", \"message\": \"" << signals[i].message << "\"}";
    }
    if (!signals.empty()) std::cout << "\n  ";
    std::cout << "],\n";
    std::cout << "  \"disclaimer\": \"Ket qua chi la ho tro tham khao, "
                 "khong xac minh danh tinh va khong bao dam giao dich an toan.\"\n";
    std::cout << "}\n";
    return 0;
}
